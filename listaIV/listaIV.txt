import random

lista = random.sample(range(1, 100), 10)
maior = lista[0]
menor = lista[0]

for numero in lista[1:]:
    if numero > maior:
        maior = numero
    if numero < menor:
        menor = numero

print(lista, maior, menor)

import random
 
 
rnum = random.sample(range(1, 100), 20)
 
par = []
impar = []
 
for i in rnum:
    if i % 2 == 0:
        par.append(i)
    else:
        impar.append(i)
 
print(f'pares: {par} \n impares: {impar} \n original: {rnum}')
