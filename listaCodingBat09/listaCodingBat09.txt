def alarm_clock(day, vacation):
  if (day == 0 or day == 6) and not vacation:
      return '10:00'
  elif (1 <= day <= 5) and not vacation:
     return '07:00' 
  elif (1 <= day <= 5) and vacation:
     return '10:00'
  elif (day == 0 or day == 6) and vacation:
     return 'off'
